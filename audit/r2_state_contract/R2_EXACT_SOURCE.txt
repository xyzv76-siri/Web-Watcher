    75	            f"multiple adapters available for target: {target.key}"

    21	class SmartFetcher:
    22	    """Polite web fetcher with ETag/Last-Modified support and timeout handling."""
    23	
    24	    def __init__(self, default_timeout: float = 10.0):
    25	        self.default_timeout = default_timeout
    26	        self.session = requests.Session()
    27	        self.session.headers.update({
    28	            "User-Agent": "WebWatcher/1.0 (https://github.com/xyzv76-siri/Web-Watcher)",
    29	            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
    30	            "Accept-Language": "en-US,en;q=0.5",
    31	            "Accept-Encoding": "gzip, deflate",
    32	            "Connection": "keep-alive",
    33	        })
    34	
    35	    def fetch(
    36	        self,
    37	        url: str,
    38	        custom_headers: Optional[Dict[str, str]] = None,
    39	        etag: Optional[str] = None,
    40	        last_modified: Optional[str] = None,
    41	        timeout: Optional[float] = None,
    42	    ) -> FetchResult:
    43	        timeout = timeout or self.default_timeout
    44	        headers = dict(self.session.headers)
    45	        if custom_headers:
    46	            headers.update(custom_headers)
    47	        if etag:
    48	            headers["If-None-Match"] = etag
    49	        if last_modified:
    50	            headers["If-Modified-Since"] = last_modified
    51	
    52	        try:
    53	            response = self.session.get(
    54	                url,
    55	                headers=headers,
    56	                timeout=timeout,
    57	                allow_redirects=True,
    58	            )
    59	            is_304 = response.status_code == 304
    60	            return FetchResult(
    61	                status_code=response.status_code,
    62	                content=response.text if not is_304 else None,
    63	                etag=response.headers.get("ETag"),
    64	                last_modified=response.headers.get("Last-Modified"),
    65	                is_304_not_modified=is_304,
    66	                error=None,
    67	                headers=dict(response.headers),
    68	            )
    69	        except RequestException as e:
    70	            return FetchResult(
    71	                status_code=None,
    72	                content=None,
    73	                etag=None,
    74	                last_modified=None,
    75	                is_304_not_modified=False,
    76	                error=str(e),
    77	                headers=None,
    78	            )
    79	
    80	    def close(self) -> None:
    81	        """Close the underlying session."""
    82	        self.session.close()
    83	
    84	    def __enter__(self):
    85	        return self
    86	
    87	    def __exit__(self, exc_type, exc_val, exc_tb):
    88	        self.close()
    89	        return False

     1	"""Single-fetch service.
     2	
     3	Orchestrates one fetch for one target:
     4	
     5	    FetchRequest
     6	        ↓
     7	    Adapter (resolved by adapter registry)
     8	        ↓
     9	    FetchResult
    10	        ↓
    11	    Persistence (FetchState upsert + optional Signal creation)
    12	
    13	The service performs exactly one fetch operation.
    14	It is NOT a scheduler, does NOT sleep, and does NOT run forever.
    15	
    16	Persistence semantics (Phase 7):
    17	
    18	    * 304 Not Modified
    19	        - does NOT replace content_hash
    20	        - does NOT create a new Signal
    21	        - preserves etag, last_modified, content_hash
    22	        - does NOT update fetched_at (existing state is authoritative)
    23	
    24	    * 200 with NEW content_hash
    25	        - upserts FetchState (etag, last_modified, content_hash, fetched_at)
    26	        - resolves target to a canonical Entity (creating it if absent)
    27	        - creates exactly one Signal with a deterministic fingerprint
    28	
    29	    * 200 with UNCHANGED content_hash
    30	        - upserts FetchState (etag, last_modified, fetched_at)
    31	        - does NOT create another Signal
    32	
    33	    * Failed FetchResult
    34	        - does NOT update content_hash
    35	        - does NOT create a new Signal
    36	        - does NOT destroy previous FetchState
    37	        - returns the failure result to the caller
    38	"""

    77	class FetchService:
    78	    """Executes a single fetch-and-persist cycle for one target."""
    79	
    80	    def __init__(
    81	        self,
    82	        repository: Repository,
    83	        adapter_registry: Optional[AdapterRegistry] = None,
    84	    ):
    85	        self._repo = repository
    86	        self._registry = adapter_registry or AdapterRegistry()
    87	
    88	    # ------------------------------------------------------------------
    89	    # Public API
    90	    # ------------------------------------------------------------------
    91	
    92	    def fetch_one(self, target: WatchTarget) -> FetchResult:
    93	        """Fetch *target* exactly once and persist the result.
    94	
    95	        Returns the FetchResult (may be success or failure).
    96	        """
    97	        adapter = self._registry.resolve(target)
    98	
    99	        # Capture prior state *before* the fetch so we can decide whether
   100	        # the new result represents a real content change.
   101	        prior = self._repo.get_fetch_state(target.key)
   102	
   103	        request = self._build_request(target, prior)
   104	        result = adapter.fetch(request)
   105	
   106	        self._apply_result(target, prior, result)
   107	
   108	        return result
   109	
   110	    # ------------------------------------------------------------------
   111	    # Persistence helpers
   112	    # ------------------------------------------------------------------
   113	
   114	    def _build_request(
   115	        self,
   116	        target: WatchTarget,
   117	        prior: Optional[FetchState],
   118	    ) -> FetchRequest:
   119	        if prior is None:
   120	            return FetchRequest(target=target)
   121	
   122	        return FetchRequest(
   123	            target=target,
   124	            etag=prior.etag,
   125	            last_modified=prior.last_modified,
   126	        )
   127	
   128	    def _apply_result(
   129	        self,
   130	        target: WatchTarget,
   131	        prior: Optional[FetchState],
   132	        result: FetchResult,
   133	    ) -> None:
   134	        """Persist a FetchResult according to the Phase 7 rules."""
   135	
   136	        if not result.success:
   137	            # Failed fetch → do NOT overwrite previous FetchState,
   138	            # do NOT create a Signal, do NOT destroy previous state.
   139	            return
   140	
   141	        # ---- 304 Not Modified ------------------------------------------------
   142	        if result.status_code == 304:
   143	            # Preserve everything: etag, last_modified, content_hash, fetched_at.
   144	            return
   145	
   146	        # ---- 200 with content ------------------------------------------------
   147	        if result.content is None or result.status_code != 200:
   148	            # No content (204 / empty) → nothing to persist as state or signal.
   149	            return
   150	
   151	        content_hash = result.content_hash or sha256_of(result.content)
   152	
   153	        # Did this observation represent a real content change?
   154	        previous_hash = prior.content_hash if prior is not None else None
   155	        new_hash = previous_hash is None or content_hash != previous_hash
   156	
   157	        # Update FetchState with fresh metadata (etag, last_modified, fetched_at)
   158	        self._repo.upsert_fetch_state(
   159	            FetchState(
   160	                target_key=result.target_key,
   161	                etag=result.etag,
   162	                last_modified=result.last_modified,
   163	                content_hash=content_hash,
   164	                fetched_at=result.fetched_at,
   165	            )
   166	        )
   167	
   168	        if not new_hash:
   169	            # Content unchanged → do NOT create another Signal.
   170	            return
   171	
   172	        # Resolve or create the canonical Entity for this target.
   173	        entity = self._repo.get_or_create_entity(
   174	            canonical_key=_canonical_entity_key(target),
   175	            name=_canonical_entity_name(target),
   176	            entity_type=_canonical_entity_type(target),
   177	        )
   178	
   179	        # Deterministic fingerprint so the same observation never
   180	        # produces duplicate Signal rows.
   181	        fingerprint = fingerprint_for_signal(
   182	            entity_id=entity.id,
   183	            signal_type=_SIGNAL_TYPE_CONTENT_CHANGE,
   184	            value=content_hash,
   185	        )
   186	
   187	        # create_signal() uses the UNIQUE(entity_id, signal_type, fingerprint)
   188	        # constraint to skip duplicates.
   189	        self._repo.create_signal(
   190	            entity_id=entity.id,
   191	            signal_type=_SIGNAL_TYPE_CONTENT_CHANGE,
   192	            observed_at=result.fetched_at,
   193	            value=content_hash,
   194	            fingerprint=fingerprint,
   195	        )

    10	def _to_utc_naive(dt: Optional[datetime]) -> Optional[datetime]:
    11	    if dt is None:
    12	        return None
    13	    if dt.tzinfo is not None:
    14	        return dt.astimezone(timezone.utc).replace(tzinfo=None)
    15	    return dt

    17	
    18	def parse_interval_seconds(interval_str: str) -> float:
    19	    s = interval_str.strip().lower()
    20	    m = re.match(r"^(\d+)([smhd])$", s)
    21	    if not m:
    22	        try:
    23	            return float(s)
    24	        except ValueError:
    25	            return 900.0  # 默认 15m
    26	    val, unit = int(m.group(1)), m.group(2)
    27	    mult = {"s": 1, "m": 60, "h": 3600, "d": 86400}
    28	    return float(val * mult[unit])
    29	
    30	
    31	def parse_retry_after(retry_after_str: Optional[str], now: Optional[datetime] = None) -> Optional[float]:
    32	    if not retry_after_str:
    33	        return None
    34	    s = retry_after_str.strip()
    35	    # 1. 秒数格式 (e.g., "120")
    36	    if s.isdigit():
    37	        return float(s)
    38	    # 2. HTTP-Date 格式 (e.g., "Wed, 21 Oct 2026 07:28:00 GMT")
    39	    try:
    40	        parsed_dt = email.utils.parsedate_to_datetime(s)
    41	        if parsed_dt:
    42	            now_dt = _to_utc_naive(now) or datetime.utcnow()
    43	            target_dt = _to_utc_naive(parsed_dt)
    44	            diff = (target_dt - now_dt).total_seconds()
    45	            return max(0.0, diff)
    46	    except Exception:
    47	        pass
    48	    return None

    72	class FetchPolicy:
    73	    """礼貌抓取与韧性决策引擎：协商缓存、退避与状态机流转"""
    74	
    75	    def __init__(
    76	        self,
    77	        base_backoff_sec: float = 30.0,
    78	        max_backoff_sec: float = 600.0,
    79	        max_consecutive_failures: int = 3,
    80	        cooldown_ladder: Optional[List[float]] = None,
    81	        jitter_ratio: float = 0.1,
    82	    ):
    83	        self.base_backoff_sec = base_backoff_sec
    84	        self.max_backoff_sec = max_backoff_sec
    85	        self.max_consecutive_failures = max_consecutive_failures
    86	        # 冷却阶梯: 30min -> 1h -> 2h -> 4h
    87	        self.cooldown_ladder = cooldown_ladder or [1800.0, 3600.0, 7200.0, 14400.0]
    88	        self.jitter_ratio = jitter_ratio

    90	    def prepare_request(self, target: Target, now: Optional[datetime] = None) -> FetchDecision:
    91	        now_dt = _to_utc_naive(now) or datetime.utcnow()
    92	        target_next = _to_utc_naive(target.next_allowed_at)
    93	
    94	        # 1. 检查是否在冷却或退避保护期内
    95	        if target_next and now_dt < target_next:
    96	            remaining = (target_next - now_dt).total_seconds()
    97	            status_val = target.status.value if hasattr(target.status, "value") else str(target.status)
    98	            return FetchDecision(
    99	                allowed=False,
   100	                delay_seconds=remaining,
   101	                reason=f"Target '{target.id}' is in {status_val} window ({int(remaining)}s remaining)",
   102	            )
   103	
   104	        # 2. 组装协商缓存头
   105	        headers: Dict[str, str] = {}
   106	        if target.etag:
   107	            headers["If-None-Match"] = target.etag
   108	        if target.last_modified:
   109	            headers["If-Modified-Since"] = target.last_modified
   110	
   111	        status_val = target.status.value if hasattr(target.status, "value") else str(target.status)
   112	        return FetchDecision(
   113	            allowed=True,
   114	            headers=headers,
   115	            reason=f"Target '{target.id}' ready for fetch (status: {status_val})",
   116	        )

   118	    def evaluate_response(
   119	        self,
   120	        target: Target,
   121	        status_code: int,
   122	        headers: Optional[Dict[str, str]] = None,
   123	        error: Optional[str] = None,
   124	        now: Optional[datetime] = None,
   125	    ) -> FetchEvaluation:
   126	        now_dt = _to_utc_naive(now) or datetime.utcnow()
   127	        headers_map = {k.lower(): v for k, v in (headers or {}).items()}
   128	        interval_sec = parse_interval_seconds(target.interval)
   129	
   130	        # A. 304 Not Modified: 协商缓存命中，直接短路
   131	        if status_code == 304:
   132	            next_allowed = now_dt + timedelta(seconds=interval_sec)
   133	            return FetchEvaluation(
   134	                target_id=target.id,
   135	                status_code=304,
   136	                new_status=TargetStatus.NORMAL,
   137	                should_emit_signal=False,
   138	                consecutive_failures=0,
   139	                next_allowed_at=next_allowed,
   140	                updated_etag=target.etag,
   141	                updated_last_modified=target.last_modified,
   142	                reason="304 Not Modified: Cache valid, no change detected",
   143	            )
   144	
   145	        # B. 200-299 Success: 正常抓取成功
   146	        if 200 <= status_code < 300:
   147	            res_etag = headers_map.get("etag") or target.etag
   148	            res_last_mod = headers_map.get("last-modified") or target.last_modified
   149	            next_allowed = now_dt + timedelta(seconds=interval_sec)
   150	            return FetchEvaluation(
   151	                target_id=target.id,
   152	                status_code=status_code,
   153	                new_status=TargetStatus.NORMAL,
   154	                should_emit_signal=True,
   155	                consecutive_failures=0,
   156	                next_allowed_at=next_allowed,
   157	                updated_etag=res_etag,
   158	                updated_last_modified=res_last_mod,
   159	                reason=f"HTTP {status_code} Success: Content fetched",
   160	            )
   161	
   162	        # C. 403 Forbidden: 触发风控或权限拒绝，直接进入长周期 COOLDOWN
   163	        if status_code == 403:
   164	            failures = target.consecutive_failures + 1
   165	            cd_sec = self.cooldown_ladder[0]
   166	            next_allowed = now_dt + timedelta(seconds=cd_sec)
   167	            return FetchEvaluation(
   168	                target_id=target.id,
   169	                status_code=403,
   170	                new_status=TargetStatus.COOLDOWN,
   171	                should_emit_signal=False,
   172	                consecutive_failures=failures,
   173	                next_allowed_at=next_allowed,
   174	                reason=f"HTTP 403 Forbidden: Entering cooldown ({int(cd_sec)}s)",
   175	            )
   176	
   177	        # D. 429 Too Many Requests / 5xx Server Error / 网络故障 (status_code == 0)
   178	        failures = target.consecutive_failures + 1
   179	        retry_after_sec = parse_retry_after(headers_map.get("retry-after"), now=now_dt)
   180	
   181	        # 连续失败达标，或在 RECOVERING/COOLDOWN 状态下探针失败 -> 进入/升级 COOLDOWN
   182	        if failures >= self.max_consecutive_failures or target.status in (TargetStatus.COOLDOWN, TargetStatus.RECOVERING):
   183	            ladder_idx = min(max(0, failures - self.max_consecutive_failures), len(self.cooldown_ladder) - 1)
   184	            cd_sec = self.cooldown_ladder[ladder_idx]
   185	            next_allowed = now_dt + timedelta(seconds=cd_sec)
   186	            return FetchEvaluation(
   187	                target_id=target.id,
   188	                status_code=status_code,
   189	                new_status=TargetStatus.COOLDOWN,
   190	                should_emit_signal=False,
   191	                consecutive_failures=failures,
   192	                next_allowed_at=next_allowed,
   193	                reason=f"Consecutive failures ({failures}) reached threshold: Escalating to cooldown ({int(cd_sec)}s)",
   194	            )
   195	
   196	        # 临时抖动 -> 进入 BACKOFF 指数退避
   197	        if retry_after_sec is not None:
   198	            delay = retry_after_sec
   199	        else:
   200	            raw_backoff = min(self.base_backoff_sec * (2 ** (failures - 1)), self.max_backoff_sec)
   201	            jitter = raw_backoff * random.uniform(-self.jitter_ratio, self.jitter_ratio)
   202	            delay = max(1.0, raw_backoff + jitter)
   203	
   204	        next_allowed = now_dt + timedelta(seconds=delay)
   205	        return FetchEvaluation(
   206	            target_id=target.id,
   207	            status_code=status_code,
   208	            new_status=TargetStatus.BACKOFF,
   209	            should_emit_signal=False,
   210	            consecutive_failures=failures,
   211	            next_allowed_at=next_allowed,
   212	            reason=f"Temporary failure (HTTP {status_code}): Backing off for {int(delay)}s",
   213	        )

     9	class DOMExtractor:
    10	    """Structured DOM/text extractor with explicit status for selector misses, empty results, and transform failures."""
    11	
    12	    @classmethod
    13	    def extract(cls, html_content: str, config: ExtractorConfig) -> ExtractionResult:
    14	        meta = {
    15	            "selector": config.selector,
    16	            "selector_type": config.selector_type,
    17	            "match_count": 0,
    18	        }
    19	
    20	        if not html_content:
    21	            return ExtractionResult(
    22	                status=ExtractionStatus.SELECTOR_NOT_FOUND,
    23	                error_message="Empty HTML content provided",
    24	                metadata=meta,
    25	            )
    26	
    27	        try:
    28	            if config.selector_type == "raw":
    29	                meta["match_count"] = 1
    30	                raw_text = html_content
    31	                return cls._apply_transforms(raw_text, config, meta)
    32	
    33	            elif config.selector_type == "regex":
    34	                matches = re.findall(config.selector, html_content)
    35	                meta["match_count"] = len(matches)
    36	                if not matches:
    37	                    return ExtractionResult(
    38	                        status=ExtractionStatus.SELECTOR_NOT_FOUND,
    39	                        error_message=f"Regex pattern '{config.selector}' matched 0 times",
    40	                        metadata=meta,
    41	                    )
    42	                raw_text = matches[0] if isinstance(matches[0], str) else matches[0][0]
    43	                return cls._apply_transforms(raw_text, config, meta)
    44	
    45	            elif config.selector_type in ("css", "xpath"):
    46	                soup = BeautifulSoup(html_content, "html.parser")
    47	                elements = soup.select(config.selector)
    48	                meta["match_count"] = len(elements)
    49	
    50	                if not elements:
    51	                    return ExtractionResult(
    52	                        status=ExtractionStatus.SELECTOR_NOT_FOUND,
    53	                        error_message=f"CSS selector '{config.selector}' found 0 matching elements",
    54	                        metadata=meta,
    55	                    )
    56	
    57	                raw_text = elements[0].get_text()
    58	                return cls._apply_transforms(raw_text, config, meta)
    59	
    60	            else:
    61	                return ExtractionResult(
    62	                    status=ExtractionStatus.TRANSFORM_ERROR,
    63	                    error_message=f"Unsupported selector_type: '{config.selector_type}'",
    64	                    metadata=meta,
    65	                )
    66	
    67	        except Exception as e:
    68	            return ExtractionResult(
    69	                status=ExtractionStatus.TRANSFORM_ERROR,
    70	                error_message=f"Extractor internal failure: {str(e)}",
    71	                metadata=meta,
    72	            )
    73	
    74	    @classmethod
    75	    def _apply_transforms(cls, raw_text: str, config: ExtractorConfig, meta: Dict[str, Any]) -> ExtractionResult:
    76	        if not config.transforms:
    77	            cleaned_val = raw_text.strip()
    78	            if not cleaned_val and raw_text != "":
    79	                return ExtractionResult(
    80	                    status=ExtractionStatus.EMPTY_AFTER_TRANSFORM,
    81	                    raw_value=raw_text,
    82	                    value="",
    83	                    metadata=meta,
    84	                )
    85	            return ExtractionResult(
    86	                status=ExtractionStatus.FOUND,
    87	                raw_value=raw_text,
    88	                value=cleaned_val,
    89	                metadata=meta,
    90	            )
    91	
    92	        try:
    93	            val = raw_text
    94	            for t_rule in config.transforms:
    95	                val = apply_transform(val, t_rule)
    96	
    97	            if isinstance(val, str) and val.strip() == "":
    98	                return ExtractionResult(
    99	                    status=ExtractionStatus.EMPTY_AFTER_TRANSFORM,
   100	                    raw_value=raw_text,
   101	                    value="",
   102	                    metadata=meta,
   103	                )
   104	
   105	            return ExtractionResult(
   106	                status=ExtractionStatus.FOUND,
   107	                raw_value=raw_text,
   108	                value=val,
   109	                metadata=meta,
   110	            )
   111	        except Exception as e:
   112	            return ExtractionResult(
   113	                status=ExtractionStatus.TRANSFORM_ERROR,
   114	                raw_value=raw_text,
   115	                value=None,
   116	                error_message=f"Transform chain failed: {str(e)}",
   117	                metadata=meta,
   118	            )

    30	class RuleEvaluator:
    31	    """网页规则评估器：执行 DOM 提取并比对变更触发条件"""
    32	
    33	    @classmethod
    34	    def evaluate_condition(cls, condition: Optional[str], old_val: Any, new_val: Any) -> bool:
    35	        if not condition:
    36	            return old_val != new_val
    37	
    38	        try:
    39	            old_num = float(old_val) if old_val is not None else 0.0
    40	            new_num = float(new_val) if new_val is not None else 0.0
    41	            delta = new_num - old_num
    42	            abs_delta = abs(delta)
    43	            pct_delta = ((new_num - old_num) / old_num * 100.0) if old_num != 0 else 0.0
    44	
    45	            context = {
    46	                "old": old_num,
    47	                "new": new_num,
    48	                "delta": delta,
    49	                "abs_delta": abs_delta,
    50	                "pct_delta": pct_delta,
    51	                "abs": abs,
    52	            }
    53	            return bool(eval(condition, {"__builtins__": None}, context))
    54	        except Exception:
    55	            return str(old_val) != str(new_val)
    56	
    57	    @classmethod
    58	    def render_template(cls, template: Optional[str], context: Dict[str, Any], default: str) -> str:
    59	        if not template:
    60	            return default
    61	        try:
    62	            return template.format(**context)
    63	        except Exception:
    64	            return template
    65	
    66	    @classmethod
    67	    def evaluate(
    68	        cls,
    69	        rule: WatcherRule,
    70	        new_html: str,
    71	        old_values: Optional[Dict[str, Any]] = None,
    72	    ) -> EvaluationResult:
    73	        old_values = old_values or {}
    74	        extracted: Dict[str, Any] = {}
    75	
    76	        # 1. 执行字段提取
    77	        for ext in rule.extractors:
    78	            result = DOMExtractor.extract(new_html, ext)
    79	            if result.status == ExtractionStatus.FOUND:
    80	                extracted[ext.name] = result.value
    81	            else:
    82	                extracted[ext.name] = None
    83	
    84	        triggered_events: List[TriggeredEvent] = []
    85	
    86	        # 2. 判定触发器
    87	        for trg in rule.triggers:
    88	            f_name = trg.field
    89	            new_val = extracted.get(f_name)
    90	            old_val = old_values.get(f_name)
    91	
    92	            is_trg = False
    93	            if old_val is not None:
    94	                if trg.type == "text_diff":
    95	                    is_trg = str(old_val) != str(new_val)
    96	                elif trg.type == "numeric_delta":
    97	                    is_trg = cls.evaluate_condition(trg.condition, old_val, new_val)
    98	                elif trg.type == "regex_match":
    99	                    if trg.condition:
   100	                        is_trg = bool(re.search(trg.condition, str(new_val)))
   101	                    else:
   102	                        is_trg = str(old_val) != str(new_val)
   103	                elif trg.type == "node_changed":
   104	                    is_trg = old_val != new_val
   105	                else:
   106	                    is_trg = old_val != new_val
   107	
   108	            if is_trg:
   109	                delta_num = 0.0
   110	                pct_num = 0.0
   111	                try:
   112	                    if old_val is not None and new_val is not None:
   113	                        delta_num = float(new_val) - float(old_val)
   114	                        pct_num = ((delta_num) / float(old_val) * 100.0) if float(old_val) != 0 else 0.0
   115	                except Exception:
   116	                    pass
   117	
   118	                tmpl_ctx = {
   119	                    "rule_name": rule.name,
   120	                    "field": f_name,
   121	                    "old_value": old_val,
   122	                    "new_value": new_val,
   123	                    "delta": delta_num,
   124	                    "abs_delta": abs(delta_num),
   125	                    "delta_percent": f"{pct_num:.2f}",
   126	                    **extracted,
   127	                }
   128	
   129	                title = cls.render_template(
   130	                    trg.title_template,
   131	                    tmpl_ctx,
   132	                    f"[{rule.name}] Field '{f_name}' changed: {old_val} -> {new_val}",
   133	                )
   134	                body = cls.render_template(
   135	                    trg.body_template,
   136	                    tmpl_ctx,
   137	                    f"Triggered by {trg.type} on '{f_name}'.\nOld: {old_val}\nNew: {new_val}",
   138	                )
   139	
   140	                triggered_events.append(TriggeredEvent(
   141	                    rule_id=rule.id,
   142	                    trigger_type=trg.type,
   143	                    field_name=f_name,
   144	                    old_value=old_val,
   145	                    new_value=new_val,
   146	                    importance=trg.importance,
   147	                    title=title,
   148	                    body=body,
   149	                    metadata={"rule_name": rule.name, "target_url": rule.target.url},
   150	                ))
   151	
   152	        return EvaluationResult(
   153	            rule_id=rule.id,
   154	            extracted_values=extracted,
   155	            triggered_events=triggered_events,
   156	            is_triggered=len(triggered_events) > 0,
   157	        )

    33	
    34	    def __init__(
    35	        self,
    36	        target: Target,
    37	        extractors: Optional[List[ExtractorConfig]] = None,
    38	        custom_headers: Optional[Dict[str, str]] = None,
    39	        timeout: float = 10.0,

    42	        self.extractors = extractors or []
    43	        self.custom_headers = custom_headers or {}
    44	        self.timeout = timeout

    46	    def execute(
    47	        self,
    48	        fetcher: Optional[SmartFetcher] = None,
    49	        policy: Optional[FetchPolicy] = None,
    50	        repo: Optional[Any] = None,
    51	        now: Optional[datetime] = None,
    52	    ) -> TargetExecutionResult:
    53	        now = now or datetime.utcnow()
    54	        fetcher = fetcher or SmartFetcher(default_timeout=self.timeout)
    55	        policy = policy or FetchPolicy()
    56	
    57	        # 1. Policy pre-check
    58	        decision = policy.prepare_request(self.target, now=now)
    59	        if not decision.allowed:
    60	            return TargetExecutionResult(
    61	                target_id=self.target.id,
    62	                allowed=False,
    63	                status_code=None,
    64	                new_status=self.target.status,
    65	                signals_emitted=[],
    66	                extracted_results={},
    67	                extracted_values={},
    68	                reason=decision.reason or "Execution skipped by policy",
    69	            )
    70	
    71	        # 2. Fetch
    72	        headers_to_send = dict(self.custom_headers)
    73	        headers_to_send.update(decision.headers)
    74	
    75	        fetch_res: FetchResult = fetcher.fetch(
    76	            url=self.target.url,
    77	            custom_headers=headers_to_send,
    78	            etag=self.target.etag,
    79	            last_modified=self.target.last_modified,
    80	            timeout=self.timeout,
    81	        )
    82	
    83	        # 3. Policy post-evaluation
    84	        headers_dict = {}
    85	        if fetch_res.etag:
    86	            headers_dict["etag"] = fetch_res.etag
    87	        if fetch_res.last_modified:
    88	            headers_dict["last-modified"] = fetch_res.last_modified
    89	
    90	        evaluation = policy.evaluate_response(
    91	            target=self.target,
    92	            status_code=fetch_res.status_code,
    93	            headers=headers_dict,
    94	            error=fetch_res.error,
    95	            now=now,
    96	        )
    97	
    98	        # 4. Update target state machine
    99	        self.target.status = evaluation.new_status
   100	        self.target.etag = evaluation.updated_etag
   101	        self.target.last_modified = evaluation.updated_last_modified
   102	        self.target.consecutive_failures = evaluation.consecutive_failures
   103	        self.target.next_allowed_at = evaluation.next_allowed_at
   104	        self.target.last_fetched_at = now
   105	
   106	        # 5. 304 short circuit
   107	        if evaluation.status_code == 304 or fetch_res.is_304_not_modified:
   108	            if repo and hasattr(repo, "save_target"):
   109	                repo.save_target(self.target)
   110	            return TargetExecutionResult(
   111	                target_id=self.target.id,
   112	                allowed=True,
   113	                status_code=304,
   114	                new_status=self.target.status,
   115	                signals_emitted=[],
   116	                extracted_results={},
   117	                extracted_values={},
   118	                is_304=True,
   119	                reason=evaluation.reason,
   120	            )
   121	
   122	        # 6. Non-success status: persist and return
   123	        if not evaluation.should_emit_signal:
   124	            if repo and hasattr(repo, "save_target"):
   125	                repo.save_target(self.target)
   126	            return TargetExecutionResult(
   127	                target_id=self.target.id,
   128	                allowed=True,
   129	                status_code=evaluation.status_code,
   130	                new_status=self.target.status,
   131	                signals_emitted=[],
   132	                extracted_results={},
   133	                extracted_values={},
   134	                reason=evaluation.reason,
   135	            )
   136	
   137	        # 7. Successful fetch: extract fields
   138	        extracted_results: Dict[str, ExtractionResult] = {}
   139	        extracted_values: Dict[str, Any] = {}
   140	        has_failures = False
   141	
   142	        for ext in self.extractors:
   143	            result = DOMExtractor.extract(fetch_res.content, ext)
   144	            extracted_results[ext.name] = result
   145	            if not result.is_found:
   146	                has_failures = True
   147	            else:
   148	                extracted_values[ext.name] = result.value
   149	
   150	        # 8. Content-hash change detection with false-positive guards
   151	        new_hash = hashlib.sha256(fetch_res.content.encode("utf-8")).hexdigest()
   152	        prev_hash = self.target.content_hash
   153	        should_emit = False
   154	        emit_reason = "Content identical, no signal emitted"
   155	
   156	        if prev_hash is None or not self.target.metadata.get("initialized"):
   157	            should_emit = True
   158	            emit_reason = "Initial fetch"
   159	        elif prev_hash != new_hash:
   160	            # Content changed; if configured extractors all failed to match, suppress as a likely false positive
   161	            if self.extractors and all(not extracted_results[e.name].is_found for e in self.extractors):
   162	                should_emit = False
   163	                emit_reason = "Content changed but all configured extractors failed; suppressing as potential false positive"
   164	            else:
   165	                should_emit = True
   166	                emit_reason = "Content changed with at least one successful extraction"
   167	        else:
   168	            should_emit = False
   169	            emit_reason = "Content identical, no signal emitted"
   170	
   171	        signals: List[Any] = []
   172	        if should_emit:
   173	            self.target.content_hash = new_hash
   174	            self.target.metadata["initialized"] = True
   175	            self.target.metadata["last_extracted"] = {k: v.value for k, v in extracted_results.items() if v.is_found}
   176	
   177	            payload = {
   178	                "target_id": self.target.id,
   179	                "url": self.target.url,
   180	                "extracted_values": extracted_values,
   181	                "extraction_results": {k: {"status": v.status.value, "value": v.value} for k, v in extracted_results.items()},
   182	                "content_hash": new_hash,
   183	                "previous_hash": prev_hash,
   184	                "status_code": fetch_res.status_code,
   185	                "captured_at": now.isoformat(),
   186	            }
   187	
   188	            sig_obj = None
   189	            if Signal is not None:
   190	                try:
   191	                    sig_obj = Signal(
   192	                        id=f"sig_{self.target.id}_{int(now.timestamp())}",
   193	                        entity_id=self.target.id,
   194	                        signal_type="WEB_CONTENT_CHANGED",
   195	                        payload=payload,
   196	                        created_at=now,
   197	                    )
   198	                except Exception:
   199	                    try:
   200	                        sig_obj = Signal(
   201	                            entity_id=self.target.id,
   202	                            signal_type="WEB_CONTENT_CHANGED",
   203	                            payload=payload,
   204	                        )
   205	                    except Exception:
   206	                        sig_obj = payload
   207	            else:
   208	                sig_obj = payload
   209	
   210	            signals.append(sig_obj)
   211	            if repo and hasattr(repo, "save_signal") and sig_obj is not None:
   212	                try:
   213	                    repo.save_signal(sig_obj)
   214	                except Exception:
   215	                    pass
   216	
   217	        # 9. Persist latest target state
   218	        if repo and hasattr(repo, "save_target"):
   219	            repo.save_target(self.target)
   220	
   221	        return TargetExecutionResult(
   222	            target_id=self.target.id,
   223	            allowed=True,
   224	            status_code=fetch_res.status_code,
   225	            new_status=self.target.status,
   226	            signals_emitted=signals,
   227	            extracted_results=extracted_results,
   228	            extracted_values=extracted_values,
   229	            is_304=False,
   230	            has_extraction_failures=has_failures,
   231	            reason=emit_reason,
   232	        )

    79	            except Exception:
    80	                try:
    81	                    return Signal(
    82	                        entity_id=self.target.id,
    83	                        signal_type=signal_type,
    84	                        payload=payload,
    85	                    )
    86	                except Exception:
    87	                    return payload
    88	        return payload
    89	
    90	    def execute(
    91	        self,
    92	        fetcher: Optional[SmartFetcher] = None,
    93	        policy: Optional[FetchPolicy] = None,
    94	        repo: Optional[Any] = None,
    95	        now: Optional[datetime] = None,
    96	    ) -> GitHubTargetExecutionResult:
    97	        now = now or datetime.utcnow()
    98	        fetcher = fetcher or SmartFetcher(default_timeout=self.timeout)
    99	        policy = policy or FetchPolicy()
   100	
   101	        # 1. 策略前置检查
   102	        decision = policy.prepare_request(self.target, now=now)
   103	        if not decision.allowed:
   104	            return GitHubTargetExecutionResult(
   105	                target_id=self.target.id,
   106	                allowed=False,
   107	                status_code=None,
   108	                new_status=self.target.status,
   109	                signals_emitted=[],
   110	                reason=decision.reason or "Skipped by policy",
   111	            )
   112	
   113	        meta = dict(self.target.metadata or {})
   114	        signals: List[Any] = []
   115	        is_any_304 = False
   116	        last_status_code = 200
   117	
   118	        # 2. 检查 Releases
   119	        if "releases" in self.watch_types:
   120	            rel_url = f"{self.BASE_API}/repos/{self.owner}/{self.repo_name}/releases/latest"
   121	            rel_etag = meta.get("release_etag")
   122	            res = fetcher.fetch(rel_url, custom_headers=self._build_headers(rel_etag), timeout=self.timeout)
   123	
   124	            eval_res = policy.evaluate_response(self.target, res.status_code, error=res.error, now=now)
   125	            self.target.status = eval_res.new_status
   126	
   127	            if res.status_code == 304 or res.is_304_not_modified:
   128	                is_any_304 = True
   129	            elif res.status_code == 200 and res.content:
   130	                try:
   131	                    rel_data = json.loads(res.content)
   132	                    tag_name = rel_data.get("tag_name")
   133	                    prev_tag = meta.get("last_release_tag")
   134	
   135	                    meta["release_etag"] = res.etag
   136	                    meta["last_release_tag"] = tag_name
   137	
   138	                    if prev_tag is not None and prev_tag != tag_name:
   139	                        sig = self._create_signal(
   140	                            "GITHUB_RELEASE_PUBLISHED",
   141	                            {
   142	                                "owner": self.owner,
   143	                                "repo": self.repo_name,
   144	                                "tag_name": tag_name,
   145	                                "release_name": rel_data.get("name"),
   146	                                "html_url": rel_data.get("html_url"),
   147	                                "published_at": rel_data.get("published_at"),
   148	                                "body": rel_data.get("body", "")[:500],
   149	                            },
   150	                            now,
   151	                        )
   152	                        signals.append(sig)
   153	                except Exception:
   154	                    pass
   155	            last_status_code = res.status_code
   156	
   157	        # 3. 检查 Repo Meta (Stars)
   158	        if "stars" in self.watch_types and self.target.status == TargetStatus.NORMAL:
   159	            repo_url = f"{self.BASE_API}/repos/{self.owner}/{self.repo_name}"
   160	            repo_etag = meta.get("repo_etag")
   161	            res = fetcher.fetch(repo_url, custom_headers=self._build_headers(repo_etag), timeout=self.timeout)
   162	
   163	            if res.status_code == 200 and res.content:
   164	                try:
   165	                    repo_data = json.loads(res.content)
   166	                    stars = repo_data.get("stargazers_count", 0)
   167	                    prev_stars = meta.get("last_stars")
   168	
   169	                    meta["repo_etag"] = res.etag
   170	                    meta["last_stars"] = stars
   171	
   172	                    if prev_stars is not None:
   173	                        delta = stars - prev_stars
   174	                        if abs(delta) >= self.star_delta_threshold:
   175	                            sig = self._create_signal(
   176	                                "GITHUB_STARS_CHANGED",
   177	                                {
   178	                                    "owner": self.owner,
   179	                                    "repo": self.repo_name,
   180	                                    "old_stars": prev_stars,
   181	                                    "new_stars": stars,
   182	                                    "delta": delta,
   183	                                },
   184	                                now,
   185	                            )
   186	                            signals.append(sig)
   187	                except Exception:
   188	                    pass
   189	            elif res.status_code == 304 or res.is_304_not_modified:
   190	                is_any_304 = True
   191	            last_status_code = res.status_code
   192	
   193	        # 4. 同步更新状态与元数据
   194	        self.target.metadata = meta
   195	        self.target.last_fetched_at = now
   196	        if repo and hasattr(repo, "save_target"):
   197	            repo.save_target(self.target)
   198	
   199	        # 5. 持久化 signals
   200	        if repo and hasattr(repo, "save_signal"):
   201	            for s in signals:
   202	                try:
   203	                    repo.save_signal(s)
   204	                except Exception:
   205	                    pass
   206	
   207	        return GitHubTargetExecutionResult(
   208	            target_id=self.target.id,
   209	            allowed=True,
   210	            status_code=last_status_code,
   211	            new_status=self.target.status,
   212	            signals_emitted=signals,
   213	            is_304=is_any_304 and len(signals) == 0,
   214	            reason=f"Emitted {len(signals)} GitHub signals" if signals else "No new events",
   215	        )

    42	class EventCorrelator:
    43	    """Correlates incoming signals into domain events and optionally triggers investigations."""
    44	
    45	    def __init__(
    46	        self,
    47	        repository: Repository,
    48	        auto_investigate: bool = False,
    49	        investigation_adapter: Optional[EventInvestigationAdapter] = None,
    50	        planner: Optional[Any] = None,
    51	        engine: Optional[Any] = None,
    52	        config: Optional[CorrelationConfig] = None,
    53	        now_factory: Optional[Any] = None,
    54	    ):
    55	        self.repository = repository
    56	        self.auto_investigate = auto_investigate
    57	        self.config = config or CorrelationConfig()
    58	        self.now_factory = now_factory or (lambda: datetime.now(timezone.utc))
    59	        self.investigation_adapter = investigation_adapter or (
    60	            EventInvestigationAdapter() if auto_investigate else None
    61	        )
    62	        self.planner = planner
    63	        self.engine = engine
    64	
    65	    def _resolve_event_type(self, signal: Signal) -> EventType:
    66	        sig_type = signal.signal_type
    67	        if sig_type == SignalType.STARS_CHANGED:
    68	            return EventType.STARS_CHANGED
    69	        if sig_type == SignalType.RELEASE_PUBLISHED:
    70	            return EventType.RELEASE_PUBLISHED
    71	        return EventType.CONTENT_CHANGE
    72	
    73	    def _evaluate_importance(self, signal: Signal, existing_event: Optional[Event] = None) -> Importance:
    74	        sig_type = signal.signal_type
    75	        if sig_type == SignalType.RELEASE_PUBLISHED:
    76	            return Importance.CRITICAL
    77	        if sig_type == SignalType.CONTENT_CHANGE:
    78	            return Importance.IMPORTANT
    79	        return Importance.INTERESTING
    80	
    81	    def process_signal(self, signal: Signal) -> Event:
    82	        """Processes an incoming signal, correlates it into an event, and triggers auto-investigation if enabled."""
    83	        entity_id = signal.entity_id
    84	        resolved_type = self._resolve_event_type(signal)
    85	        now = self.now_factory()
    86	        cutoff = now - self.config.window
    87	        open_event = self.repository.find_open_event_for_entity(
    88	            entity_id,
    89	            event_type=resolved_type,
    90	            cutoff=cutoff,
    91	        )
    92	
    93	        importance = self._evaluate_importance(signal, open_event)
    94	
    95	        if open_event is None:
    96	            event = self.repository.create_event(
    97	                entity_id=entity_id,
    98	                event_type=resolved_type,
    99	                status=EventStatus.OPEN,
   100	                importance=importance,
   101	                created_at=signal.observed_at,
   102	            )
   103	        else:
   104	            event = open_event
   105	            if (
   106	                importance == Importance.CRITICAL
   107	                or (importance == Importance.IMPORTANT and event.importance == Importance.INTERESTING)
   108	            ):
   109	                self.repository.update_event(event.id, importance=importance)
   110	                refreshed = self.repository.get_event(event.id)
   111	                if refreshed:
   112	                    event = refreshed
   113	
   114	        persisted_signal = self._persist_signal_if_needed(signal)
   115	        if persisted_signal is not None:
   116	            signal = persisted_signal
   117	
   118	        if signal.id is not None:
   119	            self.repository.link_signal_to_event(event.id, signal.id)
   120	
   121	        if self.auto_investigate:
   122	            self.dispatch_investigation(event)
   123	
   124	        return event
   125	
   126	    def _persist_signal_if_needed(self, signal: Signal) -> Optional[Signal]:
   127	        if signal.id is None:
   128	            return None
   129	        existing = self.repository.connection.execute(
   130	            "SELECT id FROM signals WHERE id = ?", (signal.id,)
   131	        ).fetchone()
   132	        if existing:
   133	            return signal
   134	        return self.repository.create_signal(
   135	            entity_id=signal.entity_id,
   136	            signal_type=signal.signal_type,
   137	            observed_at=signal.observed_at,
   138	            value=signal.value,
   139	            fingerprint=signal.fingerprint,
   140	        )
   141	
   142	    def correlate(self, signal: Signal) -> Event:
   143	        """Alias for process_signal to maintain backward compatibility."""
   144	        return self.process_signal(signal)
   145	
   146	    def close_event(self, event_id: int) -> None:
   147	        """Close an existing event so it no longer accepts new signals."""
   148	        self.repository.update_event(event_id=event_id, status=EventStatus.CLOSED)
   149	
   150	    def dispatch_investigation(self, event: Event) -> bool:
   151	        """Dispatches automatic investigation for an event if eligible and not previously investigated."""
   152	        adapter = self.investigation_adapter or EventInvestigationAdapter()
   153	        if not adapter.is_eligible(event):
   154	            return False
   155	
   156	        existing = self.repository.get_investigation_result_by_event(event.id)
   157	        if existing is not None:
   158	            return False
   159	
   160	        try:
   161	            task_type = adapter.resolve_task_type(event)
   162	            task_type_str = task_type.value if hasattr(task_type, "value") else str(task_type)
   163	
   164	            result = adapter.run_for_event(
   165	                event,
   166	                planner=self.planner,
   167	                engine=self.engine,
   168	            )
   169	            if result is None:
   170	                return False
   171	
   172	            summary = getattr(result, "summary", None) or f"Auto-investigation completed for event {event.id}"
   173	            metadata = getattr(result, "metadata", {}) or {}
   174	            raw_evidence = getattr(result, "evidence", []) or []
   175	
   176	            evidence_items = []
   177	            for item in raw_evidence:
   178	                if isinstance(item, dict):
   179	                    evidence_items.append(item)
   180	                else:
   181	                    evidence_items.append({
   182	                        "evidence_type": getattr(item, "evidence_type", "generic"),
   183	                        "payload": getattr(item, "payload", {}),
   184	                    })
   185	
   186	            inv_id = f"inv_auto_{event.id}"
   187	            status_val = "completed"
   188	            if hasattr(result, "status"):
   189	                status_val = result.status.value if hasattr(result.status, "value") else str(result.status)
   190	
   191	            self.repository.save_investigation_result(
   192	                investigation_id=inv_id,
   193	                event_id=event.id,
   194	                task_type=task_type_str,
   195	                status=status_val,
   196	                summary=summary,
   197	                metadata=metadata,
   198	                evidence_items=evidence_items,
   199	            )
   200	            return True
   201	        except Exception as exc:
   202	            logger.error(f"Auto investigation dispatch failed for event {event.id}: {exc}", exc_info=True)
   203	            return False
